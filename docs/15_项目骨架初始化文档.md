#### 1.2.1 全局异常处理器（GlobalExceptionHandler）

```java
package com.zhongjitang.common.core.exception;

import com.zhongjitang.common.core.result.R;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.http.HttpStatus;
import org.springframework.validation.FieldError;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.ResponseStatus;
import org.springframework.web.bind.annotation.RestControllerAdvice;

import javax.servlet.http.HttpServletRequest;
import java.util.stream.Collectors;

/**
 * 全局异常处理器
 * 统一处理所有Controller层抛出的异常，返回标准R<T>响应
 */
@Slf4j
@RestControllerAdvice
@RequiredArgsConstructor
public class GlobalExceptionHandler {

    /**
     * 业务异常（最常见）
     */
    @ExceptionHandler(BusinessException.class)
    @ResponseStatus(HttpStatus.OK)
    public R<Void> handleBusiness(BusinessException e, HttpServletRequest request) {
        log.warn("[业务异常] uri={}, code={}, message={}",
                request.getRequestURI(), e.getCode(), e.getMessage());
        return R.fail(e.getCode(), e.getMessage());
    }

    /**
     * 参数校验异常（@Valid触发）
     */
    @ExceptionHandler(MethodArgumentNotValidException.class)
    @ResponseStatus(HttpStatus.BAD_REQUEST)
    public R<Void> handleValidation(MethodArgumentNotValidException e, HttpServletRequest request) {
        String message = e.getBindingResult().getFieldErrors().stream()
                .map(FieldError::getDefaultMessage)
                .collect(Collectors.joining("; "));
        log.warn("[参数校验异常] uri={}, message={}", request.getRequestURI(), message);
        return R.fail(40001, message);
    }

    /**
     * 非法参数异常
     */
    @ExceptionHandler(IllegalArgumentException.class)
    @ResponseStatus(HttpStatus.BAD_REQUEST)
    public R<Void> handleIllegalArgument(IllegalArgumentException e, HttpServletRequest request) {
        log.warn("[非法参数] uri={}, message={}", request.getRequestURI(), e.getMessage());
        return R.fail(40002, e.getMessage());
    }

    /**
     * 未授权异常
     */
    @ExceptionHandler(UnauthorizedException.class)
    @ResponseStatus(HttpStatus.UNAUTHORIZED)
    public R<Void> handleUnauthorized(UnauthorizedException e, HttpServletRequest request) {
        log.warn("[未授权] uri={}, message={}", request.getRequestURI(), e.getMessage());
        return R.fail(40100, e.getMessage());
    }

    /**
     * 禁止访问异常
     */
    @ExceptionHandler(ForbiddenException.class)
    @ResponseStatus(HttpStatus.FORBIDDEN)
    public R<Void> handleForbidden(ForbiddenException e, HttpServletRequest request) {
        log.warn("[禁止访问] uri={}, message={}", request.getRequestURI(), e.getMessage());
        return R.fail(40300, e.getMessage());
    }

    /**
     * 兜底：未知系统异常
     */
    @ExceptionHandler(Exception.class)
    @ResponseStatus(HttpStatus.INTERNAL_SERVER_ERROR)
    public R<Void> handleGeneral(Exception e, HttpServletRequest request) {
        log.error("[系统异常] uri={}, message={}", request.getRequestURI(), e.getMessage(), e);
        return R.fail(50001, "系统内部错误");
    }
}
```

```java
package com.zhongjitang.common.core.result;

import com.zhongjitang.common.core.context.RequestContext;
import lombok.Data;

import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;

/**
 * 统一响应体
 * 所有API返回值必须使用R<T>包装
 */
@Data
public class R<T> {

    /** 业务状态码：0=成功，非0=失败 */
    private int code;

    /** 提示信息 */
    private String message;

    /** 业务数据 */
    private T data;

    /** 请求追踪ID */
    private String requestId;

    /** 响应时间戳 */
    private String timestamp;

    private static final DateTimeFormatter FORMATTER =
            DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm:ss");

    // ---- 成功响应 ----

    public static <T> R<T> ok() {
        return ok(null);
    }

    public static <T> R<T> ok(T data) {
        R<T> r = new R<>();
        r.setCode(0);
        r.setMessage("success");
        r.setData(data);
        r.setRequestId(RequestContext.getRequestId());
        r.setTimestamp(LocalDateTime.now().format(FORMATTER));
        return r;
    }

    public static <T> R<T> created(T data) {
        R<T> r = ok(data);
        r.setCode(201);
        r.setMessage("created");
        return r;
    }

    // ---- 失败响应 ----

    public static <T> R<T> fail(int code, String message) {
        R<T> r = new R<>();
        r.setCode(code);
        r.setMessage(message);
        r.setRequestId(RequestContext.getRequestId());
        r.setTimestamp(LocalDateTime.now().format(FORMATTER));
        return r;
    }
}
```

#### 1.2.8 JWT工具类（JwtUtil）

```java
package com.zhongjitang.common.security;

import io.jsonwebtoken.*;
import io.jsonwebtoken.security.Keys;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;

import javax.crypto.SecretKey;
import java.nio.charset.StandardCharsets;
import java.util.Date;
import java.util.Map;

/**
 * JWT工具类
 * 负责Token的生成、解析、刷新
 */
@Slf4j
@Component
public class JwtUtil {

    @Value("${jwt.secret:zhongjitang-secret-key-must-be-at-least-256-bits-long-for-hs256}")
    private String secret;

    @Value("${jwt.access-token-expiration:7200000}")
    private long accessTokenExpiration; // 默认2小时

    @Value("${jwt.refresh-token-expiration:604800000}")
    private long refreshTokenExpiration; // 默认7天

    private SecretKey getSigningKey() {
        return Keys.hmacShaKeyFor(secret.getBytes(StandardCharsets.UTF_8));
    }

    /**
     * 生成访问Token
     */
    public String generateAccessToken(Long userId, Long tenantId, Long storeId) {
        return generateToken(userId, tenantId, storeId, accessTokenExpiration);
    }

    /**
     * 生成刷新Token
     */
    public String generateRefreshToken(Long userId, Long tenantId) {
        return generateToken(userId, tenantId, null, refreshTokenExpiration);
    }

    private String generateToken(Long userId, Long tenantId, Long storeId, long expiration) {
        Date now = new Date();
        Date expiryDate = new Date(now.getTime() + expiration);

        JwtBuilder builder = Jwts.builder()
                .subject(String.valueOf(userId))
                .claim("tenant_id", tenantId)
                .claim("user_id", userId)
                .issuedAt(now)
                .expiration(expiryDate)
                .signWith(getSigningKey());

        if (storeId != null) {
            builder.claim("store_id", storeId);
        }

        return builder.compact();
    }

    /**
     * 解析Token
     */
    public Claims parseToken(String token) {
        return Jwts.parser()
                .verifyWith(getSigningKey())
                .build()
                .parseSignedClaims(token)
                .getPayload();
    }

    /**
     * 从Token中获取用户ID
     */
    public Long getUserId(String token) {
        Claims claims = parseToken(token);
        return claims.get("user_id", Long.class);
    }

    /**
     * 从Token中获取租户ID
     */
    public Long getTenantId(String token) {
        Claims claims = parseToken(token);
        return claims.get("tenant_id", Long.class);
    }

    /**
     * 从Token中获取门店ID
     */
    public Long getStoreId(String token) {
        Claims claims = parseToken(token);
        return claims.get("store_id", Long.class);
    }

    /**
     * 验证Token是否有效
     */
    public boolean validateToken(String token) {
        try {
            parseToken(token);
            return true;
        } catch (ExpiredJwtException e) {
            log.warn("[JWT] Token已过期: {}", e.getMessage());
        } catch (JwtException e) {
            log.warn("[JWT] Token无效: {}", e.getMessage());
        }
        return false;
    }

    /**
     * 判断Token是否即将过期（剩余时间<5分钟）
     */
    public boolean isTokenExpiringSoon(String token) {
        Claims claims = parseToken(token);
        Date expiration = claims.getExpiration();
        long remaining = expiration.getTime() - System.currentTimeMillis();
        return remaining < 300000; // 5分钟
    }
}
```

#### 1.2.9 多租户拦截器（TenantInterceptor）

```java
package com.zhongjitang.common.security;

import com.zhongjitang.common.core.context.TenantContext;
import com.zhongjitang.common.core.context.UserContext;
import com.zhongjitang.common.core.exception.UnauthorizedException;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Component;
import org.springframework.util.StringUtils;
import org.springframework.web.servlet.HandlerInterceptor;

import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;

/**
 * 多租户拦截器
 * 从JWT Token中解析租户ID和用户信息，设置到ThreadLocal上下文
 */
@Slf4j
@Component
@RequiredArgsConstructor
public class TenantInterceptor implements HandlerInterceptor {

    private final JwtUtil jwtUtil;

    @Override
    public boolean preHandle(HttpServletRequest request, HttpServletResponse response, Object handler) {
        String token = extractToken(request);

        if (StringUtils.hasText(token)) {
            try {
                if (jwtUtil.validateToken(token)) {
                    Long tenantId = jwtUtil.getTenantId(token);
                    Long userId = jwtUtil.getUserId(token);
                    Long storeId = jwtUtil.getStoreId(token);

                    TenantContext.setTenantId(tenantId);
                    UserContext.setUserId(userId);
                    UserContext.setStoreId(storeId);

                    log.debug("[租户拦截器] tenantId={}, userId={}", tenantId, userId);
                }
            } catch (Exception e) {
                log.warn("[租户拦截器] Token解析失败: {}", e.getMessage());
            }
        }

        return true;
    }

    @Override
    public void afterCompletion(HttpServletRequest request, HttpServletResponse response,
                                Object handler, Exception ex) {
        TenantContext.clear();
        UserContext.clear();
    }

    private String extractToken(HttpServletRequest request) {
        String bearerToken = request.getHeader("Authorization");
        if (StringUtils.hasText(bearerToken) && bearerToken.startsWith("Bearer ")) {
            return bearerToken.substring(7);
        }
        return null;
    }
}
```

```java
@Configuration

                return "tenant_id";
            }

                if (ignoreTables == null) {
                    ignoreTables = Arrays.asList(
                            "sys_user", "sys_dict", "sys_config",
                            "sys_role", "sys_permission", "sys_menu"
                    );
                }
                return ignoreTables.contains(tableName);
            }
        }));
        return interceptor;
    }
}
```

#### 1.2.11 Redis分布式锁工具类（DistributedLockUtil）

```java
package com.zhongjitang.common.redis;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.data.redis.core.StringRedisTemplate;
import org.springframework.data.redis.core.script.DefaultRedisScript;
import org.springframework.stereotype.Component;

import java.util.Collections;
import java.util.UUID;
import java.util.concurrent.TimeUnit;

/**
 * Redis分布式锁工具类
 * 基于SETNX+EXPIRE实现，Lua脚本保证原子性
 */
@Slf4j
@Component
@RequiredArgsConstructor
public class DistributedLockUtil {

    private final StringRedisTemplate redisTemplate;

    /**
     * 尝试获取分布式锁
     *
     * @param lockKey  锁的Key
     * @param expireSeconds 过期时间（秒）
     * @return 锁ID（用于释放锁），获取失败返回null
     */
    public String tryLock(String lockKey, long expireSeconds) {
        String lockId = UUID.randomUUID().toString();
        Boolean acquired = redisTemplate.opsForValue().setIfAbsent(
                lockKey, lockId, expireSeconds, TimeUnit.SECONDS);

        if (Boolean.TRUE.equals(acquired)) {
            log.debug("[分布式锁] 获取成功, key={}, lockId={}", lockKey, lockId);
            return lockId;
        }

        log.debug("[分布式锁] 获取失败, key={}", lockKey);
        return null;
    }

    /**
     * 尝试获取分布式锁（带重试）
     *
     * @param lockKey  锁的Key
     * @param expireSeconds 过期时间（秒）
     * @param retryTimes 重试次数
     * @param retryIntervalMillis 重试间隔（毫秒）
     * @return 锁ID，获取失败返回null
     */
    public String tryLockWithRetry(String lockKey, long expireSeconds,
                                   int retryTimes, long retryIntervalMillis) {
        for (int i = 0; i <= retryTimes; i++) {
            String lockId = tryLock(lockKey, expireSeconds);
            if (lockId != null) {
                return lockId;
            }
            if (i < retryTimes) {
                try {
                    Thread.sleep(retryIntervalMillis);
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                    return null;
                }
            }
        }
        return null;
    }

    /**
     * 释放分布式锁（Lua脚本保证原子性，防止误删他人锁）
     *
     * @param lockKey 锁的Key
     * @param lockId  获取锁时返回的ID
     * @return 是否释放成功
     */
    public boolean releaseLock(String lockKey, String lockId) {
        String luaScript =
                "if redis.call('get', KEYS[1]) == ARGV[1] then " +
                "   return redis.call('del', KEYS[1]) " +
                "else " +
                "   return 0 " +
                "end";

        DefaultRedisScript<Long> redisScript = new DefaultRedisScript<>(luaScript, Long.class);
        Long result = redisTemplate.execute(redisScript,
                Collections.singletonList(lockKey), lockId);

        boolean released = Long.valueOf(1L).equals(result);
        if (released) {
            log.debug("[分布式锁] 释放成功, key={}, lockId={}", lockKey, lockId);
        } else {
            log.warn("[分布式锁] 释放失败（锁已过期或被他人持有）, key={}", lockKey);
        }
        return released;
    }
}
```

#### 2.2.1 Axios请求封装（request.ts）

```typescript
// src/utils/request.ts
import axios, { type AxiosInstance, type AxiosRequestConfig, type AxiosResponse, type InternalAxiosRequestConfig } from 'axios'
import { ElMessage, ElMessageBox } from 'element-plus'
import { getToken, getRefreshToken, setToken, removeToken } from './auth'
import router from '@/router'

/** 统一响应结构 */
export interface ApiResponse<T = any> {
  code: number
  message: string
  data: T
  requestId: string
  timestamp: string
}

/** 是否正在刷新Token */
let isRefreshing = false
/** 刷新Token期间的请求队列 */
let pendingRequests: Array<(token: string) => void> = []

/** 创建Axios实例 */
const service: AxiosInstance = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL,
  timeout: 30000,
  headers: {
    'Content-Type': 'application/json'
  }
})

/** 请求拦截器 */
service.interceptors.request.use(
  (config: InternalAxiosRequestConfig) => {
    const token = getToken()
    if (token && config.headers) {
      config.headers['Authorization'] = `Bearer ${token}`
    }
    // 多租户头
    const tenantId = localStorage.getItem('tenant_id')
    if (tenantId && config.headers) {
      config.headers['X-Tenant-Id'] = tenantId
    }
    return config
  },
  (error) => {
    return Promise.reject(error)
  }
)

/** 响应拦截器 */
service.interceptors.response.use(
  (response: AxiosResponse<ApiResponse>) => {
    const res = response.data

    // code=0 表示成功
    if (res.code === 0 || res.code === 201) {
      return res as any
    }

    // Token过期，尝试刷新
    if (res.code === 40100) {
      return handleTokenExpired(response.config)
    }

    // 其他业务错误
    const errorMessage = res.message || '请求失败'
    ElMessage.error(errorMessage)
    return Promise.reject(new Error(errorMessage))
  },
  async (error) => {
    if (error.response) {
      const { status } = error.response
      switch (status) {
        case 401:
          return handleTokenExpired(error.config)
        case 403:
          ElMessage.error('无访问权限')
          break
        case 404:
          ElMessage.error('请求的资源不存在')
          break
        case 429:
          ElMessage.error('请求过于频繁，请稍后重试')
          break
        case 500:
          ElMessage.error('服务器内部错误')
          break
        default:
          ElMessage.error(error.response.data?.message || '请求失败')
      }
    } else if (error.code === 'ECONNABORTED') {
      ElMessage.error('请求超时，请稍后重试')
    } else {
      ElMessage.error('网络异常，请检查网络连接')
    }
    return Promise.reject(error)
  }
)

/** 处理Token过期 */
async function handleTokenExpired(config: AxiosRequestConfig): Promise<any> {
  const refreshToken = getRefreshToken()
  if (!refreshToken) {
    removeToken()
    router.push('/login')
    return Promise.reject(new Error('登录已过期'))
  }

  if (!isRefreshing) {
    isRefreshing = true
    try {
      const res = await axios.post(
        `${import.meta.env.VITE_API_BASE_URL}/api/v1/auth/refresh`,
        { refreshToken }
      )
      const newToken = res.data.data.accessToken
      setToken(newToken)

      // 重试队列中的请求
      pendingRequests.forEach(cb => cb(newToken))
      pendingRequests = []

      // 重试当前请求
      if (config.headers) {
        config.headers['Authorization'] = `Bearer ${newToken}`
      }
      return service(config)
    } catch (refreshError) {
      removeToken()
      router.push('/login')
      return Promise.reject(refreshError)
    } finally {
      isRefreshing = false
    }
  }

  // 正在刷新Token，加入队列等待
  return new Promise((resolve) => {
    pendingRequests.push((token: string) => {
      if (config.headers) {
        config.headers['Authorization'] = `Bearer ${token}`
      }
      resolve(service(config))
    })
  })
}

/** 通用请求方法 */
export function get<T = any>(url: string, params?: any, config?: AxiosRequestConfig): Promise<ApiResponse<T>> {
  return service.get(url, { params, ...config })
}

export function post<T = any>(url: string, data?: any, config?: AxiosRequestConfig): Promise<ApiResponse<T>> {
  return service.post(url, data, config)
}

export function put<T = any>(url: string, data?: any, config?: AxiosRequestConfig): Promise<ApiResponse<T>> {
  return service.put(url, data, config)
}

export function del<T = any>(url: string, config?: AxiosRequestConfig): Promise<ApiResponse<T>> {
  return service.delete(url, config)
}

export default service
```

```typescript
// src/router/guards.ts
import type { Router } from 'vue-router'
import { isAuthenticated } from '@/utils/auth'
import { useUserStore } from '@/stores/user'

const WHITE_LIST = ['/login', '/404', '/403']

export function setupRouterGuards(router: Router) {
  router.beforeEach(async (to, _from, next) => {
    // 白名单页面直接放行
    if (WHITE_LIST.includes(to.path)) {
      next()
      return
    }

    // 未登录跳转登录页
    if (!isAuthenticated()) {
      next(`/login?redirect=${to.fullPath}`)
      return
    }

    // 已登录，检查用户信息
    const userStore = useUserStore()
    if (!userStore.userInfo) {
      try {
        await userStore.fetchUserInfo()
      } catch {
        // 获取用户信息失败，跳转登录
        userStore.logout()
        next(`/login?redirect=${to.fullPath}`)
        return
      }
    }

    // 权限检查
    if (to.meta.roles && !to.meta.roles.some((role: string) => userStore.roles?.includes(role))) {
      next('/403')
      return
    }

    next()
  })
}
```

```typescript
    const res = await post<{
      accessToken: string
      refreshToken: string
    }>('/api/v1/auth/login', { username, password })

    setToken(res.data.accessToken)
    setRefreshToken(res.data.refreshToken)
  }

  /** 获取用户信息 */
  async function fetchUserInfo() {
    const res = await post<UserInfo>('/api/v1/auth/user-info')
    userInfo.value = res.data
    roles.value = res.data.roles
    permissions.value = res.data.permissions
  }

  /** 登出 */
  function logout() {
    userInfo.value = null
    roles.value = []
    permissions.value = []
    removeToken()
    router.push('/login')
  }

  return {
    userInfo,
    roles,
    permissions,
    login,
    fetchUserInfo,
    logout
  }
})
```

#### 3.2.1 wx.request封装（request.js）

```javascript
// utils/request.js
const { getToken, getRefreshToken, setToken, removeToken } = require('./auth')
const { envConfig } = require('./env')

/** 是否正在刷新Token */
let isRefreshing = false
/** 刷新Token期间的请求队列 */
let pendingRequests = []

/**
 * 微信小程序请求封装
 * 自动注入Token、统一错误处理、Token刷新
 */
function request(options) {
  return new Promise((resolve, reject) => {
    const token = getToken()

    const header = {
      'Content-Type': 'application/json',
      ...(options.header || {})
    }

    // 自动注入Token
    if (token) {
      header['Authorization'] = `Bearer ${token}`
    }

    // 多租户头
    const tenantId = wx.getStorageSync('tenant_id')
    if (tenantId) {
      header['X-Tenant-Id'] = tenantId
    }

    wx.request({
      url: envConfig.baseUrl + options.url,
      method: options.method || 'GET',
      data: options.data || {},
      header,
      success(res) {
        const data = res.data

        // 成功
        if (data.code === 0 || data.code === 201) {
          resolve(data)
          return
        }

        // Token过期，尝试刷新
        if (data.code === 40100) {
          handleTokenExpired(options, resolve, reject)
          return
        }

        // 其他业务错误
        wx.showToast({
          title: data.message || '请求失败',
          icon: 'none'
        })
        reject(data)
      },
      fail(err) {
        wx.showToast({
          title: '网络异常，请稍后重试',
          icon: 'none'
        })
        reject(err)
      }
    })
  })
}

/** 处理Token过期 */
function handleTokenExpired(options, resolve, reject) {
  const refreshToken = getRefreshToken()

  if (!refreshToken) {
    removeToken()
    wx.redirectTo({ url: '/pages/login/index' })
    reject(new Error('登录已过期'))
    return
  }

  if (!isRefreshing) {
    isRefreshing = true
    wx.request({
      url: envConfig.baseUrl + '/api/v1/auth/refresh',
      method: 'POST',
      data: { refreshToken },
      success(res) {
        if (res.data.code === 0) {
          const newToken = res.data.data.accessToken
          setToken(newToken)

          // 重试队列中的请求
          pendingRequests.forEach(cb => cb(newToken))
          pendingRequests = []

          // 重试当前请求
          options.header = options.header || {}
          options.header['Authorization'] = `Bearer ${newToken}`
          request(options).then(resolve).catch(reject)
        } else {
          removeToken()
          wx.redirectTo({ url: '/pages/login/index' })
          reject(new Error('Token刷新失败'))
        }
      },
      fail() {
        removeToken()
        wx.redirectTo({ url: '/pages/login/index' })
        reject(new Error('Token刷新失败'))
      },
      complete() {
        isRefreshing = false
      }
    })
  } else {
    // 正在刷新Token，加入队列等待
    pendingRequests.push((newToken) => {
      options.header = options.header || {}
      options.header['Authorization'] = `Bearer ${newToken}`
      request(options).then(resolve).catch(reject)
    })
  }
}

/** 便捷方法 */
function get(url, data) {
  return request({ url, method: 'GET', data })
}

function post(url, data) {
  return request({ url, method: 'POST', data })
}

function put(url, data) {
  return request({ url, method: 'PUT', data })
}

function del(url, data) {
  return request({ url, method: 'DELETE', data })
}

module.exports = {
  request,
  get,
  post,
  put,
  del
}
```

#### 3.2.2 微信登录流程（auth.js）

```javascript
// utils/auth.js

const { post } = require('./request')
const { envConfig } = require('./env')

const ACCESS_TOKEN_KEY = 'zjt_access_token'
const REFRESH_TOKEN_KEY = 'zjt_refresh_token'
const USER_INFO_KEY = 'zjt_user_info'

/** 获取AccessToken */
function getToken() {
  return wx.getStorageSync(ACCESS_TOKEN_KEY) || null
}

/** 设置AccessToken */
function setToken(token) {
  wx.setStorageSync(ACCESS_TOKEN_KEY, token)
}

/** 获取RefreshToken */
function getRefreshToken() {
  return wx.getStorageSync(REFRESH_TOKEN_KEY) || null
}

/** 设置RefreshToken */
function setRefreshToken(token) {
  wx.setStorageSync(REFRESH_TOKEN_KEY, token)
}

/** 清除所有Token */
function removeToken() {
  wx.removeStorageSync(ACCESS_TOKEN_KEY)
  wx.removeStorageSync(REFRESH_TOKEN_KEY)
  wx.removeStorageSync(USER_INFO_KEY)
}

/** 判断是否已登录 */
function isAuthenticated() {
  return !!getToken()
}

/**
 * 静默登录
 * 使用wx.login获取code，后端换取Token
 * 适用于：小程序启动时自动登录
 */
async function silentLogin() {
  try {
    const loginRes = await wx.login()
    if (!loginRes.code) {
      console.error('[静默登录] wx.login失败')
      return false
    }

    const res = await post('/api/v1/auth/wx-silent-login', {
      code: loginRes.code,
      appId: envConfig.appId
    })

    if (res.code === 0) {
      setToken(res.data.accessToken)
      setRefreshToken(res.data.refreshToken)
      wx.setStorageSync(USER_INFO_KEY, res.data.userInfo)
      return true
    }

    return false
  } catch (err) {
    console.error('[静默登录] 失败:', err)
    return false
  }
}

/**
 * 授权登录
 * 用户点击授权按钮后，获取手机号等信息
 * 适用于：需要用户手机号等信息的场景
 */
async function authLogin(params) {
  const { code, encryptedData, iv } = params

  const loginRes = await wx.login()

  const res = await post('/api/v1/auth/wx-auth-login', {
    code: loginRes.code,
    phoneCode: code,
    encryptedData,
    iv,
    appId: envConfig.appId
  })

  if (res.code === 0) {
    setToken(res.data.accessToken)
    setRefreshToken(res.data.refreshToken)
    wx.setStorageSync(USER_INFO_KEY, res.data.userInfo)
    return res.data
  }

  throw new Error(res.message || '授权登录失败')
}

/** 获取本地缓存的用户信息 */
function getUserInfo() {
  return wx.getStorageSync(USER_INFO_KEY) || null
}

module.exports = {
  getToken,
  setToken,
  getRefreshToken,
  setRefreshToken,
  removeToken,
  isAuthenticated,
  silentLogin,
  authLogin,
  getUserInfo
}
```

#### 3.2.4 全局状态管理（store/index.js）

```javascript
// store/index.js

/**
 * 简易全局状态管理
 * 基于wx.setStorageSync/wx.getStorageSync + 事件通知
 */

const STORE_KEY = 'zjt_store'

/** 全局状态 */
let state = wx.getStorageSync(STORE_KEY) || {
  userInfo: null,
  currentStore: null,
  cartItems: []
}

/** 监听器列表 */
const listeners = {}

/**
 * 获取状态
 */
function getState(key) {
  if (key) {
    return state[key]
  }
  return state
}

/**
 * 设置状态
 */
function setState(key, value) {
  state[key] = value
  wx.setStorageSync(STORE_KEY, state)

  // 通知监听器
  if (listeners[key]) {
    listeners[key].forEach(cb => cb(value))
  }
}

/**
 * 监听状态变化
 */
function onStateChange(key, callback) {
  if (!listeners[key]) {
    listeners[key] = []
  }
  listeners[key].push(callback)
}

/**
 * 移除监听
 */
function offStateChange(key, callback) {
  if (!listeners[key]) return
  if (callback) {
    listeners[key] = listeners[key].filter(cb => cb !== callback)
  } else {
    delete listeners[key]
  }
}

/**
 * 清除所有状态
 */
function clearState() {
  state = {
    userInfo: null,
    currentStore: null,
    cartItems: []
  }
  wx.removeStorageSync(STORE_KEY)
}

module.exports = {
  getState,
  setState,
  onStateChange,
  offStateChange,
  clearState
}
```
