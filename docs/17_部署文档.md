### 4.3 SQL版本管理规范

| 规范项 | 要求 |
|--------|------|
| 文件命名 | `V{主版本}.{次版本}.{补丁}__{描述}.sql`，如 `V1.0.0__init_base_tables.sql` |
| 版本递增 | 每次发版必须有对应的SQL文件，版本号与发版号一致 |
| 幂等性 | 使用 `IF NOT EXISTS` / `IF EXISTS`，确保可重复执行 |
| 回滚语句 | 每个增量SQL必须包含 `-- ROLLBACK:` 注释的回滚SQL |
| 审核流程 | SQL文件必须经过DBA审核后才能在生产执行 |
| 执行顺序 | 按版本号顺序执行，不可跳版本 |

### 4.4 数据库备份方案

| 环境 | 备份策略 | 保留周期 | 存储位置 |
|------|----------|----------|----------|
| dev | 手动备份 | 7天 | CloudBase对象存储 |
| test | 每日自动备份 | 30天 | CloudBase对象存储 |
| prod | 每日自动备份+实时Binlog | 90天 | 跨区域对象存储 |

#### 自动备份脚本

```bash
#!/bin/bash
# backup_db.sh - 数据库自动备份脚本

ENV=$1
DB_HOST=$2
DB_USER=$3
DB_PASS=$4
DB_NAME=$5
BACKUP_DIR="/data/backup/mysql"
DATE=$(date +%Y%m%d_%H%M%S)
BACKUP_FILE="${BACKUP_DIR}/${DB_NAME}_${DATE}.sql.gz"

# 创建备份目录
mkdir -p ${BACKUP_DIR}

# 执行备份
mysqldump -h${DB_HOST} -u${DB_USER} -p${DB_PASS} \
    --single-transaction \
    --routines \
    --triggers \
    --events \
    --set-gtid-purged=OFF \
    ${DB_NAME} | gzip > ${BACKUP_FILE}

# 检查备份是否成功
if [ $? -eq 0 ]; then
    echo "[备份成功] ${BACKUP_FILE}"
    # 计算MD5
    md5sum ${BACKUP_FILE} > ${BACKUP_FILE}.md5
else
    echo "[备份失败] ${DB_NAME}"
    exit 1
fi

# 清理过期备份
case "$ENV" in
    dev)
        KEEP_DAYS=7
        ;;
    test)
        KEEP_DAYS=30
        ;;
    prod)
        KEEP_DAYS=90
        ;;
esac

find ${BACKUP_DIR} -name "${DB_NAME}_*.sql.gz" -mtime +${KEEP_DAYS} -delete
find ${BACKUP_DIR} -name "${DB_NAME}_*.sql.gz.md5" -mtime +${KEEP_DAYS} -delete

echo "[清理完成] 保留${KEEP_DAYS}天内的备份"
```

## 五、一键启停脚本

### 5.1 部署脚本

```bash
#!/bin/bash
# deploy.sh - 一键部署脚本

set -e

ENV=$1  # dev/test/prod
VERSION=${2:-"latest"}

# 颜色输出
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m'

log_info()  { echo -e "${GREEN}[INFO]${NC} $1"; }
log_warn()  { echo -e "${YELLOW}[WARN]${NC} $1"; }
log_error() { echo -e "${RED}[ERROR]${NC} $1"; }

# 加载环境变量
load_env() {
    local env_file=".env.${ENV}"
    if [ ! -f "$env_file" ]; then
        log_error "环境配置文件不存在: $env_file"
        exit 1
    fi
    export $(grep -v '^#' $env_file | xargs)
    log_info "加载环境配置: $env_file"
}

# 前置检查
pre_check() {
    log_info "执行前置检查..."

    # 检查必要命令
    for cmd in java mvn mysql; do
        if ! command -v $cmd &> /dev/null; then
            log_error "缺少必要命令: $cmd"
            exit 1
        fi
    done

    # 检查数据库连接
    if ! mysql -h${DB_HOST} -u${DB_USER} -p${DB_PASS} -e "SELECT 1" &> /dev/null; then
        log_error "数据库连接失败"
        exit 1
    fi

    log_info "前置检查通过"
}

case "$ENV" in
    dev)
        log_info "部署开发环境..."

        # 编译打包
        mvn clean package -DskipTests

        # 启动服务
        java -jar target/zjt-api.jar \
            --spring.profiles.active=dev \
            -Xms256m -Xmx512m

        log_info "开发环境启动成功"
        ;;

    test)
        log_info "部署测试环境..."

        load_env
        pre_check

        # 编译打包
        mvn clean package -DskipTests

        # CloudBase部署
        tcb deploy --envId ${TCB_ENV_ID}

        log_info "测试环境部署成功"
        ;;

    prod)
        log_info "部署生产环境..."

        load_env
        pre_check

        # 1. 数据库备份
        log_info "备份数据库..."
        BACKUP_FILE="backup_$(date +%Y%m%d).sql"
        mysqldump -h${DB_HOST} -u${DB_USER} -p${DB_PASS} ${DB_NAME} > ${BACKUP_FILE}
        log_info "数据库备份完成: ${BACKUP_FILE}"

        # 2. 执行增量SQL
        if [ -d "sql" ] && [ "$(ls -A sql/)" ]; then
            log_warn "发现增量SQL，请确认是否执行:"
            ls -la sql/
            read -p "确认执行增量SQL? (y/n): " confirm
            if [ "$confirm" = "y" ]; then
                for sql_file in sql/V${VERSION}*.sql; do
                    if [ -f "$sql_file" ]; then
                        log_info "执行SQL: $sql_file"
                        mysql -h${DB_HOST} -u${DB_USER} -p${DB_PASS} ${DB_NAME} < $sql_file
                    fi
                done
                log_info "增量SQL执行完成"
            fi
        fi

        # 3. 编译打包
        mvn clean package -DskipTests

        # 4. CloudBase部署
        log_info "部署到CloudBase..."
        tcb deploy --envId ${TCB_ENV_ID}

        # 5. 健康检查
        log_info "等待服务启动..."
        sleep 30

        HEALTH_URL="https://zjt-api.service.tcloudbase.com/actuator/health"
        HTTP_CODE=$(curl -s -o /dev/null -w "%{http_code}" ${HEALTH_URL})

        if [ "$HTTP_CODE" = "200" ]; then
            log_info "生产环境部署成功 ✓"
        else
            log_error "健康检查失败，HTTP_CODE=${HTTP_CODE}"
            log_error "请手动检查服务状态"
            exit 1
        fi
        ;;

    *)
        echo "Usage: ./deploy.sh {dev|test|prod} [version]"
        echo ""
        echo "示例:"
        echo "  ./deploy.sh dev          # 启动开发环境"
        echo "  ./deploy.sh test         # 部署测试环境"
        echo "  ./deploy.sh prod 1.2.0   # 部署生产环境(指定版本)"
        exit 1
        ;;
esac
```

### 5.2 回滚脚本

```bash
#!/bin/bash
# rollback.sh - 一键回滚脚本

set -e

ENV=$1
TARGET_VERSION=$2

if [ -z "$ENV" ] || [ -z "$TARGET_VERSION" ]; then
    echo "Usage: ./rollback.sh {test|prod} <target_version>"
    echo ""
    echo "示例:"
    echo "  ./rollback.sh prod 1.1.0   # 回滚生产环境到1.1.0版本"
    exit 1
fi

RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m'

log_info()  { echo -e "${GREEN}[INFO]${NC} $1"; }
log_warn()  { echo -e "${YELLOW}[WARN]${NC} $1"; }
log_error() { echo -e "${RED}[ERROR]${NC} $1"; }

# 确认回滚
log_warn "即将回滚 ${ENV} 环境到版本 ${TARGET_VERSION}"
read -p "确认回滚? (y/n): " confirm
if [ "$confirm" != "y" ]; then
    log_info "取消回滚"
    exit 0
fi

case "$ENV" in
    test)
        log_info "回滚测试环境到版本 ${TARGET_VERSION}..."
        tcb rollback --envId ${TCB_TEST_ENV_ID} --version ${TARGET_VERSION}
        log_info "测试环境回滚完成"
        ;;

    prod)
        log_info "回滚生产环境到版本 ${TARGET_VERSION}..."

        # 1. CloudBase版本回退
        log_info "回退CloudBase服务版本..."
        tcb rollback --envId ${TCB_PROD_ENV_ID} --version ${TARGET_VERSION}

        # 2. 数据库回滚（如果有对应回滚SQL）
        ROLLBACK_SQL="sql/rollback/V${TARGET_VERSION}__rollback.sql"
        if [ -f "$ROLLBACK_SQL" ]; then
            log_warn "发现回滚SQL: ${ROLLBACK_SQL}"
            read -p "是否执行数据库回滚? (y/n): " db_confirm
            if [ "$db_confirm" = "y" ]; then
                mysql -h${DB_HOST} -u${DB_USER} -p${DB_PASS} ${DB_NAME} < ${ROLLBACK_SQL}
                log_info "数据库回滚完成"
            fi
        else
            log_info "未发现数据库回滚SQL，跳过数据库回滚"
        fi

        # 3. 健康检查
        sleep 30
        HEALTH_URL="https://zjt-api.service.tcloudbase.com/actuator/health"
        HTTP_CODE=$(curl -s -o /dev/null -w "%{http_code}" ${HEALTH_URL})

        if [ "$HTTP_CODE" = "200" ]; then
            log_info "生产环境回滚成功 ✓"
        else
            log_error "回滚后健康检查失败，请手动检查"
            exit 1
        fi
        ;;

    *)
        echo "Usage: ./rollback.sh {test|prod} <target_version>"
        exit 1
        ;;
esac
```

## 七、版本回滚方案

| 场景 | 回滚方式 | 操作步骤 | 预计耗时 |
|------|----------|----------|----------|
| 后端API异常 | CloudBase版本回退 | `tcb rollback --version {version}` | 5分钟 |
| 前端异常 | EdgeOne版本回退 | 重新部署上一版本 | 3分钟 |
| 数据库异常 | SQL回滚 | 执行回滚SQL | 10分钟 |
| 小程序异常 | 微信平台下架 | 小程序管理后台操作 | 30分钟 |
| 配置异常 | 环境变量回退 | 修改CloudBase环境变量+重启 | 5分钟 |
| 全局异常 | 全量回滚 | CloudBase回退+SQL回滚+前端回退 | 15分钟 |

### 7.1 回滚决策流程

```
发现线上异常
    │
    ▼
评估影响范围
    │
    ├── P0-致命(数据错误/支付故障) → 立即回滚
    │
    ├── P1-严重(核心功能不可用) → 15分钟内无法修复则回滚
    │
    ├── P2-一般(非核心功能异常) → 评估后决定，可热修复
    │
    └── P3-轻微(体验问题) → 不回滚，下版本修复
```

### 7.2 回滚注意事项

1. **数据库回滚必须谨慎**：回滚SQL可能导致数据丢失，必须先备份再执行
2. **回滚后验证**：回滚后必须执行冒烟测试，确认服务正常
3. **回滚通知**：回滚操作必须通知所有相关人员
4. **回滚记录**：记录回滚原因、操作步骤、结果，用于事后复盘
