```java
@RestController
@RequestMapping("/api/v1/{domain}/{entity}")
@RequiredArgsConstructor
@Slf4j
@Validated
public class MemberController {

    private final IMemberService memberService;

    /**
     * 分页查询会员列表
     */
    @GetMapping
    public R<PageResult<MemberResponse>> listMembers(
            @RequestParam(required = false) Long storeId,
            @RequestParam(required = false) String keyword,
            @RequestParam(defaultValue = "1") Integer page,
            @RequestParam(defaultValue = "20") Integer pageSize) {

        MemberQuery query = MemberQuery.builder()
                .storeId(storeId)
                .keyword(keyword)
                .page(page)
                .pageSize(pageSize)
                .build();

        PageResult<MemberResponse> result = memberService.pageMembers(query);
        return R.ok(result);
    }

    /**
     * 获取会员详情
     */
    @GetMapping("/{id}")
    public R<MemberDetailResponse> getMember(@PathVariable Long id) {
        MemberDetailResponse detail = memberService.getMemberDetail(id);
        return R.ok(detail);
    }

    /**
     * 创建会员
     */
    @PostMapping
    public R<MemberResponse> createMember(@Valid @RequestBody MemberCreateRequest request) {
        MemberResponse member = memberService.createMember(request);
        return R.created(member);
    }

    /**
     * 更新会员体质
     */
    @PutMapping("/{id}/constitution")
    public R<Void> updateConstitution(
            @PathVariable Long id,
            @Valid @RequestBody ConstitutionUpdateRequest request) {
        memberService.updateConstitution(id, request);
        return R.ok();
    }
}
```

```java
@Service
@RequiredArgsConstructor
@Slf4j
@Transactional(rollbackFor = Exception.class)
public class MemberServiceImpl implements IMemberService {

    private final MemberMapper memberMapper;
    private final MemberAccountMapper accountMapper;
    private final EventPublisher eventPublisher;

    @Override
    public MemberDetailResponse getMemberDetail(Long id) {
        Long tenantId = TenantContext.getTenantId(); // 多租户获取

        MemberDO member = memberMapper.selectByIdAndTenant(id, tenantId)
                .orElseThrow(() -> new MemberException(MemberErrorCode.NOT_FOUND));

        MemberDetailResponse response = convertToDetail(member);

        // 填充账户信息
        List<MemberAccountDO> accounts = accountMapper.selectByMemberId(id, tenantId);
        response.setAccounts(accounts);

        // 填充消费记录(最近10条)
        List<MemberConsumeRecord> records = memberMapper.selectRecentConsumes(id, tenantId, 10);
        response.setRecentConsumes(records);

        return response;
    }

    @Override
    @Transactional(isolation = Isolation.READ_COMMITTED)
    public MemberResponse createMember(MemberCreateRequest request) {
        Long tenantId = TenantContext.getTenantId();

        // 1. 校验手机号唯一
        if (memberMapper.existsByPhone(tenantId, request.getPhone())) {
            throw new MemberException(MemberErrorCode.PHONE_DUPLICATED);
        }

        // 2. 生成会员编号
        String memberNo = generateMemberNo(tenantId);

        // 3. 构建实体
        MemberDO member = MemberDO.builder()
                .tenantId(tenantId)
                .storeId(request.getStoreId())
                .memberNo(memberNo)
                .name(request.getName())
                .phone(request.getPhone())
                .gender(request.getGender())
                .birthday(request.getBirthday())
                .sourceChannel(request.getSourceChannel())
                .memberType(MemberType.PERSONAL.getCode())
                .status(MemberStatus.NORMAL.getCode())
                .points(0)
                .balance(BigDecimal.ZERO)
                .visitCount(0)
                .build();

        // 4. 保存
        memberMapper.insert(member);

        // 5. 创建初始账户
        createDefaultAccounts(member.getId(), tenantId);

        // 6. 发布领域事件
        eventPublisher.publish(MemberCreatedEvent.builder()
                .memberId(member.getId())
                .tenantId(tenantId)
                .sourceChannel(request.getSourceChannel())
                .build());

        return convertToResponse(member);
    }
}
```

```java
    }

// TenantInterceptor 租户拦截器
@Component
@RequiredArgsConstructor
public class TenantInterceptor implements HandlerInterceptor {
    @Override
    public boolean preHandle(HttpServletRequest request, HttpServletResponse response, Object handler) {
        String token = request.getHeader("Authorization");
        Claims claims = JwtUtil.parseToken(token);
        Long tenantId = claims.get("tenant_id", Long.class);
        TenantContext.setTenantId(tenantId);
        return true;
    }

    @Override
    public void afterCompletion(HttpServletRequest request, HttpServletResponse response, Object handler, Exception ex) {
        TenantContext.clear();
    }
}

// MyBatis Plus 多租户插件配置
@Bean
public MybatisPlusInterceptor mybatisPlusInterceptor() {
    MybatisPlusInterceptor interceptor = new MybatisPlusInterceptor();
    interceptor.addInnerInterceptor(new TenantLineInnerInterceptor(new TenantLineHandler() {
        @Override
        public Expression getTenantId() {
            return new LongValue(TenantContext.getTenantId());
        }

        @Override
        public String getTenantIdColumn() {
            return "tenant_id";

            // 系统表忽略租户
            return Arrays.asList("sys_user", "sys_dict", "sys_config").contains(tableName);
```

```java
@Data
public class R<T> {
    private int code;
    private String message;
    private T data;
    private String requestId;
    private String timestamp;

    public static <T> R<T> ok(T data) {
        R<T> r = new R<>();
        r.setCode(0);
        r.setMessage("success");
        r.setData(data);
        r.setRequestId(RequestContext.getRequestId());
        r.setTimestamp(LocalDateTime.now().format(DateTimeFormatter.ISO_LOCAL_DATE_TIME));
        return r;
    }

    public static <T> R<T> created(T data) {
        R<T> r = ok(data);
        r.setCode(201);
        r.setMessage("created");
        return r;
    }

    public static <T> R<T> fail(int code, String message) {
        R<T> r = new R<>();
        r.setCode(code);
        r.setMessage(message);
        r.setRequestId(RequestContext.getRequestId());
        r.setTimestamp(LocalDateTime.now().format(DateTimeFormatter.ISO_LOCAL_DATE_TIME));
        return r;
    }
}

// 全局异常处理
@RestControllerAdvice
@RequiredArgsConstructor
public class GlobalExceptionHandler {

    @ExceptionHandler(BusinessException.class)
    public R<Void> handleBusiness(BusinessException e) {
        log.warn("业务异常: code={}, message={}", e.getCode(), e.getMessage());
        return R.fail(e.getCode(), e.getMessage());
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public R<Void> handleValidation(MethodArgumentNotValidException e) {
        String message = e.getBindingResult().getFieldErrors().stream()
                .map(FieldError::getDefaultMessage)
                .findFirst()
                .orElse("参数校验失败");
        return R.fail(40001, message);
    }

    @ExceptionHandler(Exception.class)
    public R<Void> handleGeneral(Exception e) {
        log.error("系统异常", e);
        return R.fail(50001, "系统内部错误");
    }
}
```

```python
# app/api/v1/chat.py
from fastapi import APIRouter, Depends, HTTPException
from app.core.llm_gateway import LLMGateway
from app.models.request import ChatRequest
from app.models.response import ChatResponse
from app.api.deps import get_llm_gateway, get_current_tenant

router = APIRouter()


@router.post("/chat", response_model=ChatResponse)
async def chat(
    request: ChatRequest,
    tenant_id: str = Depends(get_current_tenant),
    llm_gateway: LLMGateway = Depends(get_llm_gateway),
) -> ChatResponse:
    """
    通用对话接口
    - task_type: chat_customer/chat_internal/content_generation
    """
    # 1. 限流检查
    await llm_gateway.check_rate_limit(tenant_id, request.task_type)

    # 2. 模型路由
    model = llm_gateway.route(request.task_type)

    # 3. 调用LLM
    result = await llm_gateway.chat(
        model=model,
        messages=request.messages,
        metadata={
            "tenant_id": tenant_id,
            "task_type": request.task_type,
        }
    )

    # 4. 计费记录
    await llm_gateway.record_usage(
        tenant_id=tenant_id,
        task_type=request.task_type,
        model=model,
        usage=result.usage,
        cost_cents=result.cost_cents,
    )

    return ChatResponse(
        request_id=result.request_id,
        model_used=model,
        latency_ms=result.latency_ms,
        usage=result.usage,
        choices=result.choices,
        cost_cents=result.cost_cents,
    )
```

### 8.1 Redis分布式锁代码模板

```java
/**
 * 预约锁档服务 - Redis分布式锁实现
 */
@Service
@RequiredArgsConstructor
@Slf4j
public class AppointmentLockService {

    private final StringRedisTemplate redisTemplate;

    private static final String LOCK_KEY_PREFIX = "appointment:lock:";
    private static final long LOCK_EXPIRE_MINUTES = 5;

    /**
     * 尝试锁定时段
     * @param storeId 门店ID
     * @param timeSlot 时段标识（如 20260604_14_15 表示14:00-15:00）
     * @param userId 用户ID
     * @return 锁定成功返回lockId，失败返回空
     */
    public Optional<String> tryLockTimeSlot(Long storeId, String timeSlot, Long userId) {
        String lockKey = LOCK_KEY_PREFIX + storeId + ":" + timeSlot;
        String lockId = UUID.randomUUID().toString();

        Boolean acquired = redisTemplate.opsForValue().setIfAbsent(
                lockKey, lockId, LOCK_EXPIRE_MINUTES, TimeUnit.MINUTES);

        if (Boolean.TRUE.equals(acquired)) {
            log.info("[预约锁档] 锁定成功, storeId={}, timeSlot={}, userId={}, lockId={}",
                    storeId, timeSlot, userId, lockId);
            return Optional.of(lockId);
        }

        log.warn("[预约锁档] 锁定失败，时段已被占用, storeId={}, timeSlot={}", storeId, timeSlot);
        return Optional.empty();
    }

    /**
     * 释放时段锁（使用Lua脚本保证原子性，防止误删他人锁）
     */
    public boolean releaseLock(Long storeId, String timeSlot, String lockId) {
        String lockKey = LOCK_KEY_PREFIX + storeId + ":" + timeSlot;

        // Lua脚本：仅当lockId匹配时才删除
        String luaScript =
                "if redis.call('get', KEYS[1]) == ARGV[1] then " +
                "   return redis.call('del', KEYS[1]) " +
                "else " +
                "   return 0 " +
                "end";

        DefaultRedisScript<Long> redisScript = new DefaultRedisScript<>(luaScript, Long.class);
        Long result = redisTemplate.execute(redisScript,
                Collections.singletonList(lockKey), lockId);

        boolean released = Long.valueOf(1L).equals(result);
        if (released) {
            log.info("[预约锁档] 释放成功, storeId={}, timeSlot={}, lockId={}", storeId, timeSlot, lockId);
        } else {
            log.warn("[预约锁档] 释放失败（锁已过期或被他人持有）, storeId={}, timeSlot={}", storeId, timeSlot);
        }
```

```java
/**
 * 防重复提交拦截器
 * 基于用户ID+请求路径+请求参数生成唯一标识，在Redis中设置短时锁
 */
@Component
@RequiredArgsConstructor
public class RepeatSubmitInterceptor implements HandlerInterceptor {

    private final StringRedisTemplate redisTemplate;

    private static final String REPEAT_KEY_PREFIX = "repeat_submit:";
    private static final long REPEAT_INTERVAL_SECONDS = 3;

    @Override
    public boolean preHandle(HttpServletRequest request, HttpServletResponse response, Object handler) {
        if (!(handler instanceof HandlerMethod)) {
            return true;
        }

        HandlerMethod handlerMethod = (HandlerMethod) handler;
        RepeatSubmit annotation = handlerMethod.getMethodAnnotation(RepeatSubmit.class);
        if (annotation == null) {
            return true;
        }

        String userId = request.getHeader("X-User-Id");
        String uri = request.getRequestURI();
        String key = REPEAT_KEY_PREFIX + userId + ":" + uri;

        Boolean acquired = redisTemplate.opsForValue().setIfAbsent(
                key, "1", REPEAT_INTERVAL_SECONDS, TimeUnit.SECONDS);

        if (Boolean.FALSE.equals(acquired)) {
            log.warn("[防重复提交] 请求过于频繁, userId={}, uri={}", userId, uri);
            throw new BusinessException(40003, "请勿重复提交");
        }

        return true;
    }
}

 * 防重复提交注解
 */

@Documented

    /**
     * 防重复间隔（秒），默认3秒
     */
```

```java
/**
 * 修改预约Service
 * 流程：校验新时段可用 → 锁定新时段 → 释放旧时段 → 更新预约 → 通知技师
 */
@Service
@RequiredArgsConstructor
@Slf4j
public class AppointmentModifyService {

    private final AppointmentMapper appointmentMapper;
    private final AppointmentLockService lockService;
    private final EventPublisher eventPublisher;

    @Transactional(rollbackFor = Exception.class)
    public AppointmentResponse modifyAppointment(Long id, AppointmentModifyRequest request) {
        Long tenantId = TenantContext.getTenantId();

        // 1. 查询预约并校验状态
        AppointmentDO appointment = appointmentMapper.selectByIdAndTenant(id, tenantId)
                .orElseThrow(() -> new AppointmentException(AppointmentErrorCode.NOT_FOUND));

        if (appointment.getStatus() != AppointmentStatus.CONFIRMED.getCode()) {
            throw new AppointmentException(AppointmentErrorCode.MODIFY_NOT_ALLOWED);
        }

        // 2. 校验新时段可用并锁定
        String newTimeSlot = buildTimeSlotKey(request.getNewStartTime());
        Optional<String> lockId = lockService.tryLockTimeSlot(
                request.getStoreId(), newTimeSlot, appointment.getMemberId());

        if (lockId.isEmpty()) {
            throw new AppointmentException(AppointmentErrorCode.TIME_SLOT_OCCUPIED);
        }

        try {
            // 3. 释放旧时段
            String oldTimeSlot = buildTimeSlotKey(appointment.getStartTime());
            lockService.releaseLock(appointment.getStoreId(), oldTimeSlot, appointment.getLockId());

            // 4. 乐观锁更新预约（version校验）
            LambdaUpdateWrapper<AppointmentDO> wrapper = new LambdaUpdateWrapper<>();
            wrapper.eq(AppointmentDO::getId, id)
                   .eq(AppointmentDO::getTenantId, tenantId)
                   .eq(AppointmentDO::getVersion, request.getVersion());

            AppointmentDO updateDO = new AppointmentDO();
            updateDO.setStartTime(request.getNewStartTime());
            updateDO.setEndTime(request.getNewEndTime());
            updateDO.setStoreId(request.getStoreId());
            updateDO.setLockId(lockId.get());
            updateDO.setVersion(appointment.getVersion() + 1);

            int rows = appointmentMapper.update(updateDO, wrapper);
            if (rows == 0) {
                throw new AppointmentException(AppointmentErrorCode.VERSION_CONFLICT);
            }

            // 5. 发布修改事件
            eventPublisher.publish(AppointmentModifiedEvent.builder()
                    .appointmentId(id)
                    .tenantId(tenantId)
                    .oldTimeSlot(oldTimeSlot)
                    .newTimeSlot(newTimeSlot)
                    .build());

            log.info("[修改预约] 成功, id={}, version={}", id, appointment.getVersion() + 1);

        } catch (Exception e) {
            // 异常时释放新时段锁
            lockService.releaseLock(request.getStoreId(), newTimeSlot, lockId.get());
            throw e;
        }

        return convertToResponse(appointment);
    }

    private String buildTimeSlotKey(LocalDateTime startTime) {
        return startTime.format(DateTimeFormatter.ofPattern("yyyyMMdd_HH_mm"));
    }
}
```

```java
/**
 * 取消预约Service
 * 流程：校验可取消 → 填写原因 → 释放时段 → 通知技师 → 更新预约状态
 */
@Service
@RequiredArgsConstructor
@Slf4j
public class AppointmentCancelService {

    private final AppointmentMapper appointmentMapper;
    private final AppointmentLockService lockService;
    private final EventPublisher eventPublisher;

    @Transactional(rollbackFor = Exception.class)
    public void cancelAppointment(Long id, AppointmentCancelRequest request) {
        Long tenantId = TenantContext.getTenantId();

        // 1. 查询预约并校验可取消
        AppointmentDO appointment = appointmentMapper.selectByIdAndTenant(id, tenantId)
                .orElseThrow(() -> new AppointmentException(AppointmentErrorCode.NOT_FOUND));

        if (appointment.getStatus() == AppointmentStatus.COMPLETED.getCode()
                || appointment.getStatus() == AppointmentStatus.IN_SERVICE.getCode()) {
            throw new AppointmentException(AppointmentErrorCode.CANCEL_NOT_ALLOWED);
        }

        // 2. 取消原因必填校验
        if (StringUtils.isBlank(request.getCancelReason())) {
            throw new AppointmentException(AppointmentErrorCode.CANCEL_REASON_REQUIRED);
        }

        // 3. 释放时段
        String timeSlot = buildTimeSlotKey(appointment.getStartTime());
        lockService.releaseLock(appointment.getStoreId(), timeSlot, appointment.getLockId());

        // 4. 更新预约状态
        appointment.setStatus(AppointmentStatus.CANCELLED.getCode());
        appointment.setCancelReason(request.getCancelReason());
        appointment.setCancelTime(LocalDateTime.now());
        appointmentMapper.updateById(appointment);

        // 5. 发布取消事件（触发退款、通知等下游流程）
        eventPublisher.publish(AppointmentCancelledEvent.builder()
                .appointmentId(id)
                .tenantId(tenantId)
                .cancelReason(request.getCancelReason())
                .cancelTime(LocalDateTime.now())
                .technicianId(appointment.getTechnicianId())
                .storeId(appointment.getStoreId())
                .build());

        log.info("[取消预约] 成功, id={}, reason={}", id, request.getCancelReason());
    }

    private String buildTimeSlotKey(LocalDateTime startTime) {
        return startTime.format(DateTimeFormatter.ofPattern("yyyyMMdd_HH_mm"));
    }
```

```java
/**
 * 推广码生成服务
 */
@Service
@RequiredArgsConstructor
@Slf4j
public class PromotionCodeService {

    private final PromotionCodeMapper promotionCodeMapper;

    /**
     * 为技师生成唯一推广码
     * 规则：ZJT + 门店编号(4位) + 技师编号(6位) + 校验位(2位)
     * 示例：ZJT0101003001AB
     */
    public PromotionCodeResponse generateCode(Long technicianId, Long storeId) {
        Long tenantId = TenantContext.getTenantId();

        // 1. 检查是否已有推广码
        Optional<PromotionCodeDO> existing = promotionCodeMapper
                .selectByTechnicianId(technicianId, tenantId);
        if (existing.isPresent()) {
            return convertToResponse(existing.get());
        }

        // 2. 生成推广码
        String storeCode = String.format("%04d", storeId % 10000);
        String techCode = String.format("%06d", technicianId % 1000000);
        String rawCode = "ZJT" + storeCode + techCode;
        String checkCode = generateCheckCode(rawCode);
        String promotionCode = rawCode + checkCode;

        // 3. 保存
        PromotionCodeDO codeDO = PromotionCodeDO.builder()
                .tenantId(tenantId)
                .technicianId(technicianId)
                .storeId(storeId)
                .promotionCode(promotionCode)
                .status(PromotionCodeStatus.ACTIVE.getCode())
                .totalShares(0)
                .totalClicks(0)
                .totalRegistrations(0)
                .totalFirstOrders(0)
                .build();

        promotionCodeMapper.insert(codeDO);
        log.info("[推广码生成] technicianId={}, code={}", technicianId, promotionCode);

        return convertToResponse(codeDO);
    }

    /**
     * 校验位生成：取rawCode各字符ASCII值加权求和，取模映射到大写字母+数字
     */
    private String generateCheckCode(String rawCode) {
        int sum = 0;
        for (int i = 0; i < rawCode.length(); i++) {
            sum += rawCode.charAt(i) * (i + 1);
        }
        String chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789";
        return String.valueOf(chars.charAt(sum % chars.length()))
                + chars.charAt((sum / chars.length()) % chars.length());
    }
}
```

### 10.3 佣金计算规则

```java
/**
 * 佣金计算服务
 */
@Service
@RequiredArgsConstructor
@Slf4j
public class CommissionService {

    private final CommissionRuleMapper ruleMapper;
    private final PromotionAttributionMapper attributionMapper;
    private final CommissionRecordMapper recordMapper;

    /**
     * 佣金比例配置
     * - 首单佣金：订单金额的 X%（默认10%）
     * - 复购佣金：订单金额的 Y%（默认5%）
     * - 佣金上限：单笔最高 Z 元（默认200元）
     */
    private static final BigDecimal FIRST_ORDER_RATE = new BigDecimal("0.10");
    private static final BigDecimal REPEAT_ORDER_RATE = new BigDecimal("0.05");
    private static final BigDecimal MAX_COMMISSION = new BigDecimal("200.00");

    /**
     * 计算并记录佣金
     * @param memberId 客户ID
     * @param orderId 订单ID
     * @param orderAmount 订单金额
     * @param isFistOrder 是否首单
     */
    @Transactional(rollbackFor = Exception.class)
    public void calculateAndRecord(Long memberId, Long orderId,
                                   BigDecimal orderAmount, boolean isFistOrder) {
        Long tenantId = TenantContext.getTenantId();

        // 1. 查找归因记录
        Optional<PromotionAttributionDO> attributionOpt =
                attributionMapper.findActiveByMemberId(memberId, tenantId);

        if (attributionOpt.isEmpty()) {
            log.debug("[佣金计算] 无归因记录，跳过, memberId={}", memberId);
            return;
        }

        PromotionAttributionDO attribution = attributionOpt.get();

        // 2. 检查归因窗口
        if (LocalDateTime.now().isAfter(attribution.getExpireAt())) {
            log.info("[佣金计算] 归因已过期, memberId={}, expireAt={}",
                    memberId, attribution.getExpireAt());
            return;
        }

        // 3. 计算佣金
        BigDecimal rate = isFistOrder ? FIRST_ORDER_RATE : REPEAT_ORDER_RATE;
        BigDecimal commission = orderAmount.multiply(rate)
                .setScale(2, RoundingMode.HALF_UP);

        // 4. 佣金上限
        if (commission.compareTo(MAX_COMMISSION) > 0) {
            commission = MAX_COMMISSION;
        }

        // 5. 记录佣金
        CommissionRecordDO record = CommissionRecordDO.builder()
                .tenantId(tenantId)
                .technicianId(attribution.getTechnicianId())
                .memberId(memberId)
                .orderId(orderId)
                .orderAmount(orderAmount)
                .commissionRate(rate)
                .commissionAmount(commission)
                .commissionType(isFistOrder ? CommissionType.FIRST_ORDER.getCode()
                        : CommissionType.REPEAT_ORDER.getCode())
                .status(CommissionStatus.PENDING.getCode())
                .build();

        recordMapper.insert(record);

        log.info("[佣金计算] 记录成功, technicianId={}, orderId={}, amount={}, commission={}",
                attribution.getTechnicianId(), orderId, orderAmount, commission);
    }
}
```
